<script>
    (function(){
        // ----- 奖品配置（一共9个） -----
        // 每个盲盒包含: 奖品名称, 对应 emoji 图标
        const PRIZE_LIST = [
            { name: '和好卷一张', emoji: '🤝' },
            { name: '奶茶卷一张', emoji: '🧋' },
            { name: '礼物卷一张', emoji: '🎁' },
            { name: '饭卷一张', emoji: '🍚' },
            { name: '要求卡一张', emoji: '📋' },
            { name: '谢谢惠顾', emoji: '🙏' },
            { name: '蛋糕卷一张', emoji: '🍰' },
            { name: '双倍卡一张', emoji: '⚡' },
            { name: '想要的东西兑换券', emoji: '💝' }
        ];

        // ----- 状态管理 -----
        let drawnCount = 0;                    // 已抽次数 (0~5)
        const MAX_DRAWS = 5;
        // 记录每个盲盒是否已被抽走 (索引 0~8)
        let boxDrawn = new Array(9).fill(false);

        // DOM 元素
        const gridContainer = document.getElementById('gridContainer');
        const remainCountSpan = document.getElementById('remainCount');
        const resetBtn = document.getElementById('resetBtn');
        const toastEl = document.getElementById('toastMessage');

        // 提示消息计时器
        let toastTimer = null;

        // ----- 工具函数 -----
        // 显示轻提示
        function showToast(msg, duration = 1800) {
            if (toastTimer) clearTimeout(toastTimer);
            toastEl.textContent = msg;
            toastEl.classList.add('show');
            toastTimer = setTimeout(() => {
                toastEl.classList.remove('show');
            }, duration);
        }

        // 更新剩余次数显示
        function updateRemainCount() {
            const left = MAX_DRAWS - drawnCount;
            remainCountSpan.textContent = left;
            if (left <= 0) {
                remainCountSpan.style.background = '#b0a09a';
            } else {
                remainCountSpan.style.background = '#c17b64';
            }
        }

        // 根据当前状态重新渲染九宫格 (保留原始奖品信息，追加已抽状态)
        function renderGrid() {
            // 清空但保留结构
            gridContainer.innerHTML = '';

            for (let i = 0; i < PRIZE_LIST.length; i++) {
                const prize = PRIZE_LIST[i];
                const boxDiv = document.createElement('div');
                boxDiv.className = 'box';
                if (boxDrawn[i]) {
                    boxDiv.classList.add('disabled');
                }

                // 盲盒内部结构
                // 如果已被抽走，显示奖品名称 + 对勾，否则显示神秘"❓"和"盲盒"
                if (boxDrawn[i]) {
                    // 已抽走：展示奖品具体内容 + 勾选标记
                    boxDiv.innerHTML = `
                        <div class="emoji-big">${prize.emoji}</div>
                        <div class="prize-name">${prize.name}</div>
                        <div class="check-mark">✓</div>
                    `;
                } else {
                    // 未抽走：展示盲盒样式
                    boxDiv.innerHTML = `
                        <div class="emoji-big">❓</div>
                        <div class="prize-name">盲盒</div>
                    `;
                }

                // 添加点击事件 (只有未抽走 && 还有次数时才触发)
                boxDiv.addEventListener('click', (e) => {
                    e.stopPropagation();
                    handleBoxClick(i);
                });

                gridContainer.appendChild(boxDiv);
            }
        }

        // 处理点击盲盒
        function handleBoxClick(index) {
            // 1. 检查是否已抽过该盒子
            if (boxDrawn[index]) {
                showToast('这个盲盒已经被抽走啦～');
                return;
            }

            // 2. 检查抽奖次数是否已达上限
            if (drawnCount >= MAX_DRAWS) {
                showToast('你已经抽满5次啦，点击"重新开始"再玩吧');
                return;
            }

            // 3. 执行抽奖
            // 标记该盒子已抽
            boxDrawn[index] = true;
            drawnCount++;

            // 获取奖品信息
            const prize = PRIZE_LIST[index];

            // 重新渲染网格（更新盒子显示和禁用状态）
            renderGrid();
            updateRemainCount();

            // 显示抽中结果
            showToast(`✨ 抽到了：${prize.name} ${prize.emoji}`);

            // 检查是否所有次数用尽
            if (drawnCount >= MAX_DRAWS) {
                // 稍微延迟提示，让用户看到抽奖结果
                setTimeout(() => {
                    showToast('🎉 五次机会已用完，点"重新开始"再来一局吧', 2500);
                }, 600);
            }
        }

        // 重置游戏
        function resetGame() {
            drawnCount = 0;
            boxDrawn = new Array(9).fill(false);
            renderGrid();
            updateRemainCount();
            showToast('🔄 盲盒已全部重置！可以重新抽取啦');
        }

        // 初始化 / 刷新整个界面
        function init() {
            drawnCount = 0;
            boxDrawn = new Array(9).fill(false);
            renderGrid();
            updateRemainCount();

            // 绑定重置按钮事件
            resetBtn.addEventListener('click', resetGame);
        }

        // 启动
        init();

        // 针对微信/移动端优化：防止点击时页面缩放（已在 meta 中处理）
        // 并且消除双击缩放 (添加 touch-action)
        document.querySelectorAll('.box, .reset-btn').forEach(el => {
            el.style.touchAction = 'manipulation';
        });

    })();
</script>
